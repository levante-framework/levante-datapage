function interval(range = [], options = {}) {
  const [min = 0, max = 1] = range;
  const {
    step = .001,
    label = null,
    value = [min, max],
    format = ([start, end]) => `${start} … ${end}`,
    color,
    width,
    theme,
    __ns__ = randomScope(),
  } = options;

  const css = `
#${__ns__} {
  //font: 13px/1.2 var(--sans-serif);
  display: flex;
  align-items: baseline;
  flex-wrap: wrap;
  max-width: 100%;
  width: auto;
  flex-direction: column;
}
@media only screen and (min-width: 30em) {
  #${__ns__} {
    flex-wrap: nowrap;
    width: 360px;
  }
}
#${__ns__} .label {
  //width: 120px;
  //padding: 5px 0 4px 0;
  //padding: 5px 0 0 0;
  //margin-right: 6.5px;
  flex-shrink: 0;
}
#${__ns__} .form {
  display: flex;
  width: 100%;
}
#${__ns__} .range {
  flex-shrink: 1;
  width: 100%;
}
#${__ns__} .range-slider {
  width: 100%;
  margin-bottom: .3em;
  margin-top: .3em;
}
  `;
  
  const $range = rangeInput({min, max, value: [value[0], value[1]], step, color, width, theme});
  const $output = html`<output>`;
  const $view = html`<div id=${__ns__}>
${label == null ? '' : html`<div class="label">${label}`}
<div class=form>
  <div class=range>
    ${$range}<div class=range-output>${$output}</div>
  </div>
</div>
${html`<style>${css}`}
  `;

  const update = () => {
    const content = format([$range.value[0], $range.value[1]]);
    if(typeof content === 'string') $output.value = content;
    else {
      while($output.lastChild) $output.lastChild.remove();
      $output.appendChild(content);
    }
  };
  $range.oninput = update;
  // the slider only fires input on release, so follow its drags for the readout
  $range.addEventListener('dragging', update);
  update();
  
  return Object.defineProperty($view, 'value', {
    get: () => $range.value,
    set: ([a, b]) => {
      $range.value = [a, b];
      update();
    },
  });
}

// date version of interval. the underlying input only handles numbers, so work
// in epoch milliseconds internally and report the value as dates. step is in
// days, and the range is widened to whole UTC days so values land on dates.
function dateInterval(range = [], options = {}) {
  const day = 864e5;
  const fmtDate = d3.utcFormat("%Y-%m-%d");
  const [min, max] = [d3.utcDay.floor(range[0]), d3.utcDay.ceil(range[1])];
  const {
    step = 1,
    value = [min, max],
    format = ([start, end]) => `${fmtDate(start)} … ${fmtDate(end)}`,
    ...rest
  } = options;

  const $interval = interval([+min, +max], {
    ...rest,
    step: step * day,
    // widen outward, else the slider snaps to the nearest day and clips the ends
    value: [+d3.utcDay.floor(value[0]), +d3.utcDay.ceil(value[1])],
    format: ([start, end]) => format([new Date(start), new Date(end)]),
  });

  // input events bubble up from the slider, so the wrapper works as a viewof
  const $view = html`<div>${$interval}`;
  return Object.defineProperty($view, 'value', {
    get: () => $interval.value.map(ms => new Date(ms)),
    set: ([a, b]) => {
      $interval.value = [+a, +b];
    },
  });
}

function rangeInput(options = {}) {
  const {
    min = 0,
    max = 100,
    step = 'any',
    value: defaultValue = [min, max],
    color,
    width,
    theme = theme_Flat,
  } = options;
  
  const controls = {};
  const scope = randomScope();
  const clamp = (a, b, v) => v < a ? a : v > b ? b : v;

  // Will be used to sanitize values while avoiding floating point issues.
  const input = html`<input type=range ${{min, max, step}}>`;
  
  const dom = html`<div class=${`${scope} range-slider`} style=${{
    color,
    width: cssLength(width),
  }}>
  ${controls.track = html`<div class="range-track">
    ${controls.zone = html`<div class="range-track-zone">
      ${controls.range = html`<div class="range-select" tabindex=0>
        ${controls.min = html`<div class="thumb thumb-min" tabindex=0>`}
        ${controls.max = html`<div class="thumb thumb-max" tabindex=0>`}
      `}
    `}
  `}
  ${html`<style>${theme.replace(/:scope\b/g, '.'+scope)}`}
</div>`;

  let value = [], changed = false, dragging = false;
  Object.defineProperty(dom, 'value', {
    get: () => [...value],
    set: ([a, b]) => {
      value = sanitize(a, b);
      updateRange();
    },
  });

  const sanitize = (a, b) => {
    a = isNaN(a) ? min : ((input.value = a), input.valueAsNumber);
    b = isNaN(b) ? max : ((input.value = b), input.valueAsNumber);
    return [Math.min(a, b), Math.max(a, b)];
  }
  
  const updateRange = () => {
    const ratio = v => (v - min) / (max - min);
    dom.style.setProperty('--range-min', `${ratio(value[0]) * 100}%`);
    dom.style.setProperty('--range-max', `${ratio(value[1]) * 100}%`);
  };

  const dispatch = name => {
    dom.dispatchEvent(new Event(name, {bubbles: true}));
  };
  const setValue = (vmin, vmax) => {
    const [pmin, pmax] = value;
    value = sanitize(vmin, vmax);
    updateRange();
    // Only dispatch if values have changed.
    if(pmin === value[0] && pmax === value[1]) return;
    // mid-drag this only moves the readout; 'input' (which downstream cells
    // recompute on) fires once on release rather than on every step crossed
    dispatch(dragging ? 'dragging' : 'input');
    changed = true;
  };
  
  setValue(...defaultValue);
  
  // Mousemove handlers.
  const handlers = new Map([
    [controls.min, (dt, ov) => {
      const v = clamp(min, ov[1], ov[0] + dt * (max - min));
      setValue(v, ov[1]);
    }],
    [controls.max, (dt, ov) => {
      const v = clamp(ov[0], max, ov[1] + dt * (max - min));
      setValue(ov[0], v);
    }],
    [controls.range, (dt, ov) => {
      const d = ov[1] - ov[0];
      const v = clamp(min, max - d, ov[0] + dt * (max - min));
      setValue(v, v + d);
    }],
  ]);
  
  // Returns client offset object.
  const pointer = e => e.touches ? e.touches[0] : e;
  // Note: Chrome defaults "passive" for touch events to true.
  const on  = (e, fn) => e.split(' ').map(e => document.addEventListener(e, fn, {passive: false}));
  const off = (e, fn) => e.split(' ').map(e => document.removeEventListener(e, fn, {passive: false}));
  
  let initialX, initialV, target;
  function handleDrag(e) {
    // Gracefully handle exit and reentry of the viewport.
    if(!e.buttons && !e.touches) {
      handleDragStop();
      return;
    }
    dragging = true;
    const w = controls.zone.getBoundingClientRect().width;
    e.preventDefault();
    handlers.get(target)((pointer(e).clientX - initialX) / w, initialV);
  }
  
  
  function handleDragStop(e) {
    off('mousemove touchmove', handleDrag);
    off('mouseup touchend', handleDragStop);
    if(changed) {
      dispatch('input');
      dispatch('change');
    }
  }
  
  invalidation.then(handleDragStop);
  
  dom.ontouchstart = dom.onmousedown = e => {
    dragging = false;
    changed = false;
    if(!handlers.has(e.target)) return;
    on('mousemove touchmove', handleDrag);
    on('mouseup touchend', handleDragStop);
    e.preventDefault();
    e.stopPropagation();
    
    target = e.target;
    initialX = pointer(e).clientX;
    initialV = value.slice();
  };
  
  controls.track.onclick = e => {
    if(dragging) return;
    changed = false;
    const r = controls.zone.getBoundingClientRect();
    const t = clamp(0, 1, (pointer(e).clientX - r.left) / r.width);
    const v = min + t * (max - min);
    const [vmin, vmax] = value, d = vmax - vmin;
    if(v < vmin) setValue(v, v + d);
    else if(v > vmax) setValue(v - d, v);
    if(changed) dispatch('change');
  };
  
  return dom;
}

function randomScope(prefix = 'scope-') {
  return prefix + (performance.now() + Math.random()).toString(32).replace('.', '-');
}

// keeps an input's value in a URL parameter, so that links to the page can
// share its state. on creation the input is set from the parameter (if valid),
// and the parameter is updated whenever the input changes, with it left out
// when the input is at its default value
//
// choices: Map of option labels to values for radio/select/checkbox inputs,
// which are stored by label (their values can be null or data), comma separated
// for checkboxes; otherwise the value is stored as a number, or as comma
// separated numbers for intervals
url_bound = (key, input, { choices } = {}) => {
  const is_number = s => s !== "" && Number.isFinite(+s)
  const label_of = v => [...choices].find(([label, value]) => value === v)?.[0]
  const encode = choices && Array.isArray(input.value)
    ? v => v.map(label_of).join(",")
    : choices
    ? label_of
    : v => String(v)
  const decode = choices && Array.isArray(input.value)
    ? s => { const v = s === "" ? [] : s.split(","); return v.every(l => choices.has(l)) ? { value: v.map(l => choices.get(l)) } : null }
    : choices
    ? s => choices.has(s) ? { value: choices.get(s) } : null
    : Array.isArray(input.value)
    ? s => { const v = s.split(","); return v.length === 2 && v.every(is_number) ? { value: v.map(Number) } : null }
    : s => is_number(s) ? { value: +s } : null

  const default_param = encode(input.value)
  const param = new URLSearchParams(location.search).get(key)
  const decoded = param === null ? null : decode(param)
  if (decoded !== null) {
    input.value = decoded.value
    // the value setter alone is silent, so let the input's own listeners (e.g.
    // slider fill) know
    input.dispatchEvent(new Event("input", { bubbles: true }))
  }

  input.addEventListener("input", () => {
    const url = new URL(location.href)
    const p = encode(input.value)
    if (p === undefined || p === default_param) url.searchParams.delete(key)
    else url.searchParams.set(key, p)
    history.replaceState(history.state, "", url)
  })
  return input
}

// pearson correlation of two arrays
cor = (x, y) => {
  const mx = d3.mean(x), my = d3.mean(y)
  const sxy = d3.sum(x, (xi, i) => (xi - mx) * (y[i] - my))
  return sxy / Math.sqrt(d3.sum(x, xi => (xi - mx) ** 2) * d3.sum(y, yi => (yi - my) ** 2))
}

// 95% confidence interval of a correlation r over n observations, via fisher's z
cor_ci = (r, n) => {
  if (!(n > 3)) return [NaN, NaN]
  const z = Math.atanh(r), h = 1.96 / Math.sqrt(n - 3)
  return [Math.tanh(z - h), Math.tanh(z + h)]
}

// empirical reliability = var(score) / (var(score) + mean(se^2)), from arrays
// of scores and squared ses
rxx = (s, e2) => { const v = d3.variance(s); return v / (v + d3.mean(e2)) }

// 95% percentile bootstrap interval of empirical reliability, resampling runs
// (written out with typed arrays since it runs for every group on every change)
rxx_ci = (s, e2, rand, B = 500) => {
  const n = s.length
  if (n < 2 || e2.some(isNaN)) return [NaN, NaN]
  const out = new Float64Array(B)
  for (let b = 0; b < B; b++) {
    let sum = 0, sum2 = 0, sum_e2 = 0
    for (let i = 0; i < n; i++) {
      const j = Math.floor(rand() * n)
      sum += s[j]; sum2 += s[j] ** 2; sum_e2 += e2[j]
    }
    const v = (sum2 - sum ** 2 / n) / (n - 1)
    out[b] = v / (v + sum_e2 / n)
  }
  return [d3.quantile(out, 0.025), d3.quantile(out, 0.975)]
}

// least squares slope of y on x and its standard error
ols_slope = (x, y) => {
  const n = x.length, mx = d3.mean(x), my = d3.mean(y)
  const sxx = d3.sum(x, xi => (xi - mx) ** 2)
  const b = d3.sum(x, (xi, i) => (xi - mx) * (y[i] - my)) / sxx
  const ss_res = d3.sum(x, (xi, i) => (y[i] - my - b * (xi - mx)) ** 2)
  return { b, se: Math.sqrt(ss_res / (n - 2) / sxx) }
}

// reliability, age correlation, and age slope, with 95% intervals, for each
// dataset x task x scoring model. the bootstrap's random numbers are seeded, so that the same
// inputs always give the same intervals
score_stats = (rs) => {
  const rand = d3.randomLcg(42)
  return d3.flatRollup(rs, g => {
    const s = Float64Array.from(g, d => d.score)
    const e2 = Float64Array.from(g, d => d.score_se == null ? NaN : d.score_se ** 2)
    const r_age = cor(g.map(d => d.score), g.map(d => d.age))
    const [rxx_lo, rxx_hi] = rxx_ci(s, e2, rand)
    const [age_lo, age_hi] = cor_ci(r_age, g.length)
    // slope of scores in sds of their scoring model's scores, with normal interval
    const { task_id, task_code, scoring_model } = g[0]
    const sd = score_sds.get(task_id).get(task_code).get(scoring_model)
    const { b, se } = ols_slope(g.map(d => d.age), g.map(d => d.score / sd))
    return {
      n_scores: g.length,
      empirical_rxx: rxx(s, e2), rxx_lo, rxx_hi,
      cor_score_age: r_age, age_lo, age_hi,
      slope_age: b, slope_se: se, slope_lo: b - 1.96 * se, slope_hi: b + 1.96 * se,
      age_sd: d3.deviation(g, d => d.age)
    }
  }, d => d.dataset, d => d.task_id, d => d.task_code, d => d.scoring_model)
  .map(([dataset, task_id, task_code, scoring_model, s]) => ({ dataset, task_id, task_code, scoring_model, ...s, Task: task_info[task_code] }))
}

// test-retest correlation, with 95% interval, for each dataset x task
retest_stats = (ps) => d3.flatRollup(ps, g => {
    const trt = cor(g.map(d => d.score_1), g.map(d => d.score_2))
    const [trt_lo, trt_hi] = cor_ci(trt, g.length)
    return { n_pairs: g.length, trt, trt_lo, trt_hi, gap_median: d3.median(g, d => d.gap) }
  }, d => d.dataset, d => d.task_id, d => d.task_code)
  .map(([dataset, task_id, task_code, s]) => ({ dataset, task_id, task_code, ...s, Task: task_info[task_code] }))

// weighted average of a statistic over the datasets shown for each task -- for
// correlations, averaged as fisher's z (with weights n - 3, this is a fixed
// effect meta-analysis)
pooled = (rows, x, weight, { fisher = false } = {}) => d3.flatRollup(
  rows.filter(d => Number.isFinite(d[x]) && weight(d) > 0),
  g => {
    const w = g.map(weight)
    const v = g.map(d => fisher ? Math.atanh(d[x]) : d[x])
    const m = d3.sum(v, (vi, i) => vi * w[i]) / d3.sum(w)
    return fisher ? Math.tanh(m) : m
  }, d => d.Task)
  .map(([Task, value]) => ({ Task, value }))

// order of tasks for every plot's y axis and colors -- by highest reliability
// over all runs (so that it doesn't change with the inputs), with tasks that
// have none last
task_order = d3.groupSort(score_stats(runs),

NoteAbout these statistics

Each dot is one dataset, sized proportional to its number of scores, with a line for its 95% confidence interval. The black tick is the average across the datasets shown. Scores without an age are left out, and the first per child option uses only each child’s earliest assessment on a task within the age range.

Empirical reliability is var(θ) / (var(θ) + mean(SE2)) over the ability scores θ and their standard errors. Its confidence interval is from a bootstrap of scores (500 resamples); the average weights datasets by number of scores. Sentence Reading scores have no standard errors, so no reliability.

Age correlation is the Pearson correlation of scores with age. Its confidence interval is from Fisher’s z, and the average is of Fisher’s z, weighted by N − 3.

Note that reliability and age correlation both depend on how spread out ages are in a dataset: a narrower age range gives lower values.

Age slope is the least squares slope of scores on age, with scores divided by the SD of all scores from their scoring model. So the units are score SDs per year, and it doesn’t depend on a dataset’s age range. It’s comparable across datasets that share a scoring model, but less so across tasks or scoring models, since each one’s SD depends on the ages and datasets it has scores for. Its confidence interval is ±1.96 standard errors, and the average is weighted by inverse variance.

NoteAbout these statistics

Each dot is one dataset, sized by its number of pairs, with a line for its 95% interval (from Fisher’s z). The black tick is the average across the datasets shown, of Fisher’s z weighted by N − 3.

A pair is two consecutive assessments of the same child on the same task, ordered by start time: 1→2 is their first and second assessments, 2→3 their second and third. The retest gap is the time between the two assessments’ starts, and the age is the child’s age at the first assessment.

Test-retest correlation is the Pearson correlation of the two assessments’ scores. Over longer gaps it reflects children’s development (stability of their rank order) as well as the reliability of measurement.

  console.log('getting scores')

}
